  Widget build(BuildContext context) {
    final state = context.stepsCubit.stateValue;
    
    return Scaffold(
      appBar: BaktazAppBar(title: context.l10n.stepsTitle),
      body: state.isLoading
          ? const Shimmer.loading()
          : state.error != null
              ? BaktazErrorCard(
                  error: state.error!,

              : _buildContent(context, state),
    );
  }
}

class StepChecklistWidget extends StatelessWidget {
  const StepChecklistWidget({
    super.key,
    required this.diagnostic,
  });

  final StepConnectionDiagnostic diagnostic;

  @override
  Widget build(BuildContext context) {
    return BaktazCard(
      child: Padding(
        padding: Paddings.medium,
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            BaktazText(context.l10n.stepsChecklistTitle, style: AppTextStyle.titleSmall),
            Gap(AppSizes.small),
            _buildItem(context, context.l10n.stepsServiceAvailable, diagnostic != StepConnectionDiagnostic.notInstalled),
            _buildItem(context, context.l10n.stepsPermissionGranted, diagnostic == StepConnectionDiagnostic.authorized || diagnostic == StepConnectionDiagnostic.connected),
            _buildItem(context, context.l10n.stepsDataAccessible, diagnostic == StepConnectionDiagnostic.connected),
            _buildItem(context, context.l10n.stepsRecentDataAvailable, diagnostic == StepConnectionDiagnostic.connected),
          ],
        ),
      ),
    );
  }

  Widget _buildItem(BuildContext context, String label, bool isOk) {
    return Row(
      children: [
        Icon(isOk ? Icons.check_circle : Icons.cancel, color: isOk ? Colors.green : Colors.red, size: 18),
        Gap(AppSizes.small),
        BaktazText(label, style: AppTextStyle.bodyMedium),
      ],
    );
  }
}

```dart
class StepSyncStatusTile extends StatelessWidget {
  const StepSyncStatusTile({
    super.key,
    required this.status,
    required this.lastSyncAt,
  });

  final StepSyncStatus status;
  final DateTime? lastSyncAt;

  @override
  Widget build(BuildContext context) {
    return BaktazCard(
      child: Padding(
        padding: Paddings.medium,
        child: Row(
          children: [
            _statusIcon(status),
            Gap(AppSizes.medium),
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  BaktazText(_statusTitle(status, context), style: AppTextStyle.titleSmall),
                  BaktazText(_statusSubtitle(status, lastSyncAt, context), style: AppTextStyle.bodySmall),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _statusIcon(StepSyncStatus status) {
    switch (status) {
      case StepSyncStatus.synced:
        return const Icon(Icons.check_circle, color: Colors.green);
      case StepSyncStatus.error:
        return const Icon(Icons.error, color: Colors.red);
      case StepSyncStatus.noData:
        return const Icon(Icons.warning, color: Colors.orange);
      default:
        return const Icon(Icons.sync, color: Colors.grey);
    }
  }

  String _statusTitle(StepSyncStatus status, BuildContext context) {
    switch (status) {
      case StepSyncStatus.synced:
        return context.l10n.stepsSyncStatusSynced;
      case StepSyncStatus.error:
        return context.l10n.stepsSyncStatusError;
      case StepSyncStatus.noData:
        return context.l10n.stepsSyncStatusNoData;
      default:
        return context.l10n.stepsSyncStatusPending;
    }
  }

  String _statusSubtitle(HealthSyncStatus status, DateTime? lastSyncAt, BuildContext context) {
    if (lastSyncAt == null) return context.l10n.stepsNeverSynced;
    return context.l10n.stepsLastSyncTime(time: lastSyncAt.toLocalString(context));
  }
}
```
