```dart
class HealthPage extends StatelessWidget {
  const HealthPage({super.key});

  @override
  Widget build(BuildContext context) {
    return BlocSignalProvider(
      child: const HealthPageView(),
    );
  }
}

class HealthPageView extends StatelessWidget {
  const HealthPageView({super.key});

  @override
  Widget build(BuildContext context) {
    final state = context.healthCubit.stateValue;
    
    return UnfocusableScaffold(
      appBar: AppAppBar(title: context.l10n.healthTitle),
      body: state.isLoading
          ? const Shimmer.loading()
          : state.error != null
              ? BaktazErrorScreen(
                  error: state.error!,
                  onRetry: () => context.healthCubit.loadConnectionStatus(),
                )
              : _buildContent(context, state),
    );
  }
}
```

Content layout:
- **Diagnostic checklist (`HealthChecklistWidget`) is shown on every visit**, providing 4-line status: ✓ Health service available, ✓ Permission granted, ✓ Step data accessible, ✓ Recent data available. For empty-state users, checklist shows ✕ markers. For connected users, it shows ✓ markers with current stats.
- List of connected devices (`HealthConnectionCard` for each)
- Connection status banner at top (`HealthSyncStatusTile`)
- "Connect Health" CTA button when no integration exists
- Disconnect option on each connected device card

### Task 3: Create Health Sync Status Tile

**Files:**
- Create: `lib/features/health/presentation/widgets/health_sync_status_tile.dart`

- [ ] **Step 1: Implement `HealthSyncStatusTile`**

```dart
class HealthSyncStatusTile extends StatelessWidget {
  const HealthSyncStatusTile({
    super.key,
    required this.status,
    required this.lastSyncAt,
  });

  final HealthSyncStatus status;
  final DateTime? lastSyncAt;

  @override
  Widget build(BuildContext context) {
    return BaktazCard(
      child: Padding(
        padding: Paddings.medium,
        child: Row(
          children: [
            _statusIcon(status),
            Gap(AppSizes.medium),
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  BaktazText(_statusTitle(status), style: AppTextStyle.titleSmall),
                  BaktazText(_statusSubtitle(status, lastSyncAt), style: AppTextStyle.bodySmall),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _statusIcon(HealthSyncStatus status) {
    switch (status) {
      case HealthSyncStatus.synced:
        return const Icon(Icons.check_circle, color: Colors.green);
      case HealthSyncStatus.error:
        return const Icon(Icons.error, color: Colors.red);
      case HealthSyncStatus.noData:
        return const Icon(Icons.warning, color: Colors.orange);
      default:
        return const Icon(Icons.sync, color: Colors.grey);
    }
  }
}
```

### Task 5: Update Home Daily Step Hero Card with Adaptive Button

**Files:**
- Modify: `lib/features/home/presentation/widgets/home_daily_step_hero_card.dart`

- [ ] **Step 1: Add adaptive button based on connection state**

> **Per grilling decision (Option A):** Reuse existing sync button — behavior changes based on connection state.
> - When **connected**: triggers step sync (calls `HealthRepository.getTodaySteps()` + `StepsRepository.syncSteps()`)
> - When **NOT connected**: shows "Connect Health" CTA instead of refresh button

```dart
// In HomeDailyStepHeroCard.build():
Widget _buildActionButton(BuildContext context, HomeState state) {
  if (state.isConnected) {
    // Connected: trigger step sync inline
    return HomeDailyStepSyncFooter(
      onSyncTap: () async {
        final healthCubit = context.read<HealthCubit>();
        final stepsCubit = context.read<StepsCubit>();
        await healthCubit.getTodaySteps();
        await stepsCubit.syncSteps();
      },
    );
  }
  
  // Not connected: show "Connect Health" CTA → navigates to /account/health
  return HealthConnectCTABanner(
    onConnect: () => context.go('/account/health'),
  );
}
```

Note: When connected, sync happens inline without navigation. When not connected, user is navigated to `/account/health` to establish connection. After connection from home, user stays on HealthPage (no bounce-back).
